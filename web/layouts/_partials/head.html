{{- /* Everything a search engine and a link preview read about a page. A page with no description stops the build: the description is what a search result shows under the title, and an empty one is filled in by the engine from whatever text it finds first. */ -}} {{- $title := or .Params.seoTitle .Title }} {{- $description := .Description }} {{- /* A search result shows about sixty characters of a title. The name of the program goes after the page's own title only when both fit, so the words that say what the page is about are never the ones cut off. */ -}} {{- /* A title that already names the program is not given the name twice. */ -}} {{- $branded := printf "%s - %s" $title site.Title }} {{- $named := strings.Contains (lower $title) (lower site.Title) }} {{- if and (not .IsHome) (not $named) (le (strings.RuneCount $branded) site.Params.titleLimit) }}{{ $title = $branded }}{{ end }} {{- if and (not $description) (ne .Kind "404") }} {{- errorf "%s has no description, and a search result would make one up" .File.Path }} {{- end }} {{ $title }} {{- if eq .Kind "404" }} {{- else }} {{- range .AllTranslations }} {{- end }} {{- /* English is the default language, and the address for every other reader. */ -}} {{- range where .AllTranslations "Language.Lang" "en" }} {{- end }} {{- end }} {{- $css := resources.Get "css/site.css" | minify | fingerprint }} {{- $icon := resources.Get "icon.png" }} {{- $favicon := $icon.Resize "64x64 png" }} {{- $touch := $icon.Resize "180x180 png" }} {{- $picture := resources.Get "social-preview.png" }} {{- range .Translations }} {{- end }} {{ partial "structured-data.html" . }}
