# A name with A records and no AAAA.
#
# WRITTEN FROM THE ZONE FILE, BEFORE certdoctor WAS EVER POINTED AT THIS BENCH.
# That order is the whole value of this file - a verdict written after a run is a
# transcript of what the program does, and it will agree with the program however
# wrong the program is.

target=https://only4.ok.test/

# The zone declares one A record and no AAAA at all. Asking for AAAA therefore
# gets NOERROR with an empty answer section - NODATA - which is a different fact
# from "no such name" and is one of the three certdoctor queries resolvers
# directly in order to tell apart.
direct_a=NOERROR
direct_aaaa=NODATA

# The union of the direct view is one address, and the platform resolving the
# same name through the same server can only arrive at the same one.
addresses=192.0.2.10
l1_status=PASS
divergence=agree

# Never these: both lenses answered and both found the same address, so there is
# nothing disjoint about them and neither of them came back empty.
divergence_forbidden=disjoint,agree_none,one_sided

# Six layers are still unwritten and report UNKNOWN, so the run cannot be
# conclusive whatever L1 says.
exit_code=4

# L3 connects to what this name resolved to. The zone hands out 192.0.2.10, from
# the range RFC 5737 reserves for documentation - so nothing anywhere routes it,
# and the attempt cannot succeed no matter whose machine runs this bench.
#
# WHICH FAILURE IT IS depends on the network this runs on: a machine that knows
# it has no route says so at once, one that does not simply waits. Both are
# honest, so the expectation names the pair rather than pinning the one that
# happens to come out here.
l3_status=FAIL
l3_outcome=timeout|unreachable

# THE EXIT CODE MOVED WHEN L3 ARRIVED, and the expectation says why rather than
# being quietly corrected: every address in this zone is from a documentation
# range, so nothing routes and L3 reports FAIL. A failing layer makes the run a
# diagnosis, whatever the unwritten layers say.
